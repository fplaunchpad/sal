\section{The Verification Problem}
% ===========================================================================================

\begin{frame}
  \frametitle{Why Verification Matters}
  
  \begin{quote}
    \textit{Designing correct replicated data types is challenging.}
  \end{quote}
  
  \vspace{0.3cm}
  Even well-known designs have subtle bugs:
  \begin{itemize}
    \item RGA (Replicated Growable Arrays) anomaly discovered after publication
    \item Enable-wins flag bug
    \item Ordering anomalies in complex CRDTs
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Replication-Aware (RA) Linearizability}
  
  \begin{block}{Definition}
    RA linearizability relates replica executions to a sequential explanation of updates, ensuring that:
    \begin{itemize}
      \item Merge semantics are explicit
      \item Linearizability-style reading is maintained
      \item Conflict resolution is respected
    \end{itemize}
  \end{block}
  
  \vspace{0.3cm}
  Neem reduced RA-linearizability to 24 verifiable conditions (VCs)
\end{frame}

\section{Verification with F* and its Limitations}
% ===========================================================================================

\begin{frame}
  \frametitle{Prior Work: Neem in \fstar}
  
  \begin{block}{Strengths}
    \begin{itemize}
      \item SMT-aided automation for RA linearizability
      \item Formal reduction to 24 verification conditions
      \item Practical verification of CRDTs and MRDTs
    \end{itemize}
  \end{block}
  
  \vspace{0.3cm}
  \begin{block}{The Problem: Trust in SMT Solvers}
    \begin{itemize}
      \item SMT-centric workflows are opaque when automation fails
      \item Failed VC provides little actionable feedback
      \item Relies on external solver $\rightarrow$ larger trusted computing base (TCB)
      \item Debugging requires manual proof-debugging workflow
      \item SMT proofs are brittle to small changes
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{The Debugging Problem}
  
  \textbf{Traditional SMT-aided workflow:}
  \begin{enumerate}
    \item VC fails to be discharged
    \item Add intermediate assertions (\texttt{assert})
    \item Progressively push assertions deeper
    \item Hope to localize the failing reasoning step
    \item Perhaps adjust SMT-solver hints (unfolding, limits, etc.)
  \end{enumerate}
  
  \vspace{0.3cm}
  \textbf{Compare to:} Regular software development
  \begin{itemize}
    \item Write code $\rightarrow$ Run tests $\rightarrow$ Inspect failing traces $\rightarrow$ Refine
    \item Fast iteration and clear feedback
  \end{itemize}
\end{frame}

    \sal is more than tactics—it's a collaborative verification workflow:

    \item \textbf{SMT when needed:} For harder goals

  \textit{Think: Agentic system with human-in-the-loop}

\begin{frame}
  \frametitle{AI-Assisted Theorem Proving}
  
  \textbf{When kernel and SMT automation aren't enough:}
  
  \begin{itemize}
    \item Provide LLM with proof context
    \item LLM suggests tactics or intermediate steps
    \item Developer reviews and accepts/refines suggestions
    \item Fallback mechanism: always has a working proof
  \end{itemize}
  
  \vspace{0.3cm}
  \textbf{Advantages:}
  \begin{itemize}
    \item Reduces programmer burden on truly difficult proofs
    \item Maintains small TCB (kernel-verified proofs)
    \item Preserves human understanding
  \end{itemize}
\end{frame}

      \item If it fails: get concrete counterexample
      \item Visualize execution $\rightarrow$ see what's wrong
      \item Fix and iterate (like running tests!)
    \end{itemize}
  \end{block}

  \textbf{Key requirement:} VCs must be decidable propositions

    \item Plausible generates random inputs, checks membership

  \begin{align*}
    \mu(e_1(0), e_3(e_1(0)), e_1(e_4(e_2(0)))) &= \\
    e_3(\mu(e_1(0), e_1(0), e_1(e_4(e_2(0)))))
  \end{align*}
  
  \small Can be evaluated directly when all operations are computable!
\end{frame}

    \item Visualization shows the story
    \item Developers understand root cause immediately
